To reproduce the Linux comparison without emitting checkout paths, this Python 3 probe reports package names and versions as one JSON record. Its hostile-value assertion covers a newline, ESC and a quote. The edge selection follows the normal-closure check at `tools/plugin-conformance/isolation.py:20`.

```python
import json
import subprocess

def closure(flags):
    args = ["cargo", "tree", "--offline", "--locked", "-p", "dctx-cli",
            "--target", "x86_64-unknown-linux-gnu", "-e", "normal",
            "--prefix", "none", "--color", "never", *flags]
    run = subprocess.run(args, capture_output=True, text=True)
    if run.returncode:
        raise SystemExit(record(run.stderr))
    return {" ".join(row.split()[:2]) for row in run.stdout.splitlines()}

def record(value):
    return json.dumps(value, ensure_ascii=True)

hostile = 'crate\n\x1b"'
assert record(hostile) == r'"crate\n\u001b\""'
default = closure([])
strict = closure(["--no-default-features", "--features", "strict"])
print(record({"default": len(default), "strict": len(strict),
              "default_only": sorted(default - strict), "strict_only": sorted(strict - default)}))
```
